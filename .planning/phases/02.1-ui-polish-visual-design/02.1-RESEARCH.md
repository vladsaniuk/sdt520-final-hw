export const CodeSnippet: React.FC<CodeSnippetProps> = ({ code, language }) => {
  const [copied, setCopied] = useState(false)

          {language.toUpperCase()}
        </span>

          onClick={handleCopy}
          className="text-xs text-gray-400 hover:text-aws-orange transition-colors font-sans"
        >
          {copied ? 'Copied!' : 'Copy code'}
        </button>
      </div>

        language={language}
        style={vscDarkPlus}
        customStyle={{
          margin: 0,
          padding: '1rem',
          fontFamily: "'Geist Mono', monospace",
          fontSize: '0.75rem',
        }}
      >
        {code}
      </SyntaxHighlighter>
    </div>
  )
}

### Anti-Patterns to Avoid

- **Do not use `module.exports` in tailwind.config.js** — project uses Tailwind v4 with `export default`
- **Do not add new color tokens** — all needed colors already exist (`aws-orange`, `aws-squid`, `aws-squid-dark`, `aws-squid-light`, `aws-orange-dark`)
- **Do not push error into `messages` array** — error is a separate UI layer (see Pattern 7)
- **Do not auto-send on chip click** — chip click only fills textarea (`setInput(chip)`)
- **Do not change mermaid.initialize config** — `theme: 'default'`, `startOnLoad: false`, `securityLevel: 'loose'` must stay as-is
- **Do not add shadcn, Radix, or icon packages** — UI-SPEC mandates raw Tailwind + inline SVG only
- **Do not add CDN font links** — use @fontsource npm packages only

## Common Pitfalls

### Pitfall 1: Geist Font Name Mismatch

**What goes wrong:** Font family declared as `'Geist Variable'` or `'GeistVF'` instead of `'Geist'`
**Why it happens:** @fontsource registers the family name exactly as `'Geist'` — the variable font name from Vercel's site differs
**How to avoid:** Use exactly `font-family: 'Geist', system-ui, -apple-system, sans-serif` in CSS; use `font-family: "'Geist Mono', monospace"` in SyntaxHighlighter customStyle
**Warning signs:** Body text falls back to system-ui silently — no console error

### Pitfall 2: mermaid.css Font Override Persists

**What goes wrong:** `frontend/src/styles/mermaid.css` still declares `font-family: 'Inter', sans-serif` on `.mermaid` — diagram nodes use Inter instead of Geist
**Why it happens:** mermaid.css is loaded in main.tsx and applies scoped overrides to `.mermaid` elements
**How to avoid:** Update line 2 of mermaid.css: `font-family: 'Geist', sans-serif;`
**Warning signs:** Diagram text looks different from body text

### Pitfall 3: Tailwind v4 Config Format Confusion

**What goes wrong:** Changing `export default` to `module.exports` breaks the build
**Why it happens:** v3 tutorials use CommonJS format; this project is v4 with ESM
**How to avoid:** Never touch tailwind.config.js format — only add to `theme.extend.colors` if needed (no additions needed this phase)
**Warning signs:** PostCSS build error mentioning `module.exports` or unexpected token

### Pitfall 4: CostTable Wrapper Double-Wrapping

**What goes wrong:** CostTable renders its own `bg-white rounded-lg` wrapper AND ChatBox wraps it in another `bg-white border rounded-xl` container
**Why it happens:** ChatBox (already in place) wraps CostTable in a panel; CostTable v1 also had its own wrapper
**How to avoid:** Per UI-SPEC — CostTable's new wrapper is `rounded-xl overflow-hidden border border-gray-200 shadow-sm my-2`. ChatBox wraps CostTable in its own panel. Look at how ChatBox calls `<CostTable>` — it's inside a `bg-white border border-gray-200 rounded-xl shadow-sm overflow-hidden` panel. This means CostTable's `my-2` wrapper and `border` would double-border. **Solution:** Remove the outer CostTable container div and let the table itself be the root element within the `<div className="p-4">` in ChatBox — OR keep the wrapper but omit border since parent provides it. The UI-SPEC says "Wrapper: `rounded-xl overflow-hidden border border-gray-200 shadow-sm my-2` (remove outer bg-white/rounded-lg — wrapper handles it)" — this means the wrapper STAYS, and ChatBox's panel wrapping is removed for the cost section.
**Warning signs:** Double border visible around cost table

### Pitfall 5: DEMO_MESSAGES Still Loaded on Refresh

**What goes wrong:** Comment out DEMO_MESSAGES but forget to change the useState initializer
**Why it happens:** DEMO_MESSAGES constant is defined but initializer still passes it
**How to avoid:** Change `useState<Message[]>(DEMO_MESSAGES)` → `useState<Message[]>([])`. The DEMO_MESSAGES const can remain defined (useful for testing) but must not be the initial state.

### Pitfall 6: Error Banner Placement in JSX Tree

**What goes wrong:** Error banner renders inside the scrollable messages area, gets scrolled away
**Why it happens:** Placing `{error && ...}` inside the `flex-1 overflow-y-auto` div
**How to avoid:** Error banner renders in the `shrink-0` bottom section — ABOVE the input bar but OUTSIDE the scrollable area. Place it between the scroll area and the `<div className="shrink-0 bg-white border-t ...">` input bar.
